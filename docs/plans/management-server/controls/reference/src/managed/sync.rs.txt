//! Policy sync: ask the management server what changed and act on its answer.

use std::sync::atomic::{AtomicBool, AtomicU64, Ordering};
use std::sync::Arc;
use std::time::Duration;

use probation_policy::{ChangesRequest, ChangesResponse};
use reqwest::header::CONTENT_TYPE;

use super::cache::{CompiledInterner, KeyCache, LayerCache};
use super::lock::NetworkLock;

pub struct KeySync {
    http: reqwest::Client,
    changes_url: String,
    timeout: Duration,
    cache: Arc<KeyCache>,
    layers: Arc<LayerCache>,
    compiled: Arc<CompiledInterner>,
    network_lock: Arc<NetworkLock>,
    retired: Arc<AtomicBool>,
    cursor: AtomicU64,
}

/// What one round of `changes(since)` came back with.
enum Answer {
    Changes(ChangesResponse),
    /// The cursor is older than the retained change log: the whole answer, with the full network-lock table.
    Reset(ChangesResponse),
    Retired,
    /// A 4xx other than 401, 403 and 429: the server refused this request (for example an oversized batch).
    Rejected,
}

/// What the caller does with the activity batch it sent.
pub enum Synced {
    /// The answer was applied; the batch was delivered.
    Applied,
    /// The server refused the batch; count it as dropped, do not re-queue it.
    BatchRejected,
}

/// The server could not be reached (connect error, timeout, 5xx, 429, or 403 `FIREWALL_SOURCE_REFUSED`); the caller re-queues the batch.
pub struct Unreachable;

struct Failed;

impl KeySync {
    /// One sync round. Only a server answer may shrink the key cache.
    pub async fn fetch_changes(&self, request: &ChangesRequest) -> Result<Synced, Unreachable> {
        let answer = match self.request_changes(request).await {
            Ok(answer) => answer,
            Err(_) => return Err(Unreachable),
        };
        self.mark_reachable();
        match answer {
            Answer::Changes(delta) => self.apply_changes(&delta),
            Answer::Reset(full) => {
                self.reset_cursor(full.revision);
                self.cache.raise_invalidation_floor(full.revision);
                self.drop_cache();
                self.network_lock.adopt(full.network_lock.as_deref());
            }
            Answer::Retired => self.stop_serving(),
            Answer::Rejected => return Ok(Synced::BatchRejected),
        }
        Ok(Synced::Applied)
    }

    async fn request_changes(&self, request: &ChangesRequest) -> Result<Answer, Failed> {
        let body = serde_json::to_vec(request).map_err(|_| Failed)?;
        let sending = self.http.post(&self.changes_url).header(CONTENT_TYPE, "application/json").body(body).send();
        let response = tokio::time::timeout(self.timeout, sending).await.map_err(|_| Failed)?.map_err(|_| Failed)?;
        let status = response.status();
        if status == reqwest::StatusCode::UNAUTHORIZED {
            return Ok(Answer::Retired);
        }
        if status == reqwest::StatusCode::TOO_MANY_REQUESTS || source_refused(&response) {
            return Err(Failed);
        }
        if status.is_client_error() {
            return Ok(Answer::Rejected);
        }
        if !status.is_success() {
            return Err(Failed);
        }
        let bytes = response.bytes().await.map_err(|_| Failed)?;
        let changes: ChangesResponse = serde_json::from_slice(&bytes).map_err(|_| Failed)?;
        if changes.retired {
            return Ok(Answer::Retired);
        }
        if changes.reset {
            return Ok(Answer::Reset(changes));
        }
        Ok(Answer::Changes(changes))
    }

    fn mark_reachable(&self) {
        self.cache.note_server_seen();
    }

    fn reset_cursor(&self, revision: u64) {
        self.cursor.store(revision, Ordering::Release);
    }

    /// Adopt a sent network-lock table, mark changed layers stale, update changed rate limits in place,
    /// remove changed keys, expire changed places, then move the cursor past the delta. Each
    /// invalidation records `delta.revision`, so a late answer read before it is dropped.
    fn apply_changes(&self, delta: &ChangesResponse) {
        self.network_lock.adopt(delta.network_lock.as_deref());
        self.layers.mark_stale(&delta.changed_layers, delta.revision);
        self.cache.update_limits(&delta.changed_limits, delta.revision);
        self.cache.remove_keys(&delta.changed_keys, delta.revision);
        self.cache.expire_places(&delta.changed_places, delta.revision);
        self.advance_cursor(delta.revision);
    }

    fn advance_cursor(&self, revision: u64) {
        self.cursor.store(revision, Ordering::Release);
    }

    /// Everything the firewall learned from the change log: keys, layers and compiled policies.
    fn drop_cache(&self) {
        self.cache.clear();
        self.layers.clear();
        self.compiled.clear();
    }

    /// The server retired this firewall or rejected its token: serve nothing more.
    fn stop_serving(&self) {
        self.retired.store(true, Ordering::Release);
        self.cache.clear();
    }
}

/// 403 with the `FIREWALL_SOURCE_REFUSED` code: the server refused this firewall's source address. Handled like an
/// unreachable server; any other 403 is a rejected request.
fn source_refused(response: &reqwest::Response) -> bool {
    response.status() == reqwest::StatusCode::FORBIDDEN
        && response.headers().get("x-probation-error").is_some_and(|code| code == "FIREWALL_SOURCE_REFUSED")
}
